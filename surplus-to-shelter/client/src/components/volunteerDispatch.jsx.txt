import React from 'react';
import { Navigation, Phone, CheckCircle, Truck } from 'lucide-react';

export default function VolunteerDispatch({ items = [], onAdvanceStatus }) {
  const tasks = items.filter(i => i.status !== 'POSTED' && i.status !== 'EXPIRED_ROUTED');

  const getNextAction = (status) => {
    if (status === 'MATCHED') return { label: 'Accept Task', next: 'PICKUP_ASSIGNED' };
    if (status === 'PICKUP_ASSIGNED') return { label: 'Confirm Picked Up', next: 'IN_TRANSIT' };
    if (status === 'IN_TRANSIT') return { label: 'Confirm Delivered', next: 'DELIVERED' };
    return null;
  };

  return (
    <div className="space-y-4">
      <div className="border-b pb-3">
        <h2 className="text-xl font-bold text-slate-900">Volunteer Dispatch & Transit Tasks</h2>
        <p className="text-xs text-slate-500">Pick up matched batches and execute verified shelter handovers.</p>
      </div>

      {tasks.length === 0 ? (
        <div className="p-8 text-center bg-white rounded-2xl border text-slate-500 text-sm">
          No dispatch tasks active. When shelters accept food, tasks appear here.
        </div>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          {tasks.map(task => {
            const action = getNextAction(task.status);
            const mapsUrl = `https://www.google.com/maps/dir/?api=1&origin=${encodeURIComponent(task.pickupAddress)}&destination=Shelter+Main+Hub`;

            return (
              <div key={task.id} className="bg-white p-5 rounded-2xl border border-slate-200 shadow-sm space-y-3">
                <div className="flex justify-between items-center">
                  <span className="text-xs font-mono font-bold text-slate-700">Task #{task.id}</span>
                  <span className="text-[11px] font-bold px-2.5 py-0.5 rounded-full bg-emerald-100 text-emerald-800">
                    {task.status}
                  </span>
                </div>

                <div>
                  <h4 className="font-bold text-slate-900">{task.foodName} ({task.servings} Servings)</h4>
                  <p className="text-xs text-slate-500">Pickup: {task.pickupAddress}</p>
                </div>

                <div className="flex gap-2 pt-1">
                  <a
                    href={mapsUrl}
                    target="_blank"
                    rel="noreferrer"
                    className="flex-1 py-2 px-3 rounded-lg border border-slate-300 hover:bg-slate-50 text-slate-700 text-xs font-bold flex items-center justify-center gap-1"
                  >
                    <Navigation className="w-3.5 h-3.5 text-blue-600" /> Open Maps
                  </a>
                  <a
                    href={`tel:${task.contactPhone}`}
                    className="flex-1 py-2 px-3 rounded-lg border border-slate-300 hover:bg-slate-50 text-slate-700 text-xs font-bold flex items-center justify-center gap-1"
                  >
                    <Phone className="w-3.5 h-3.5 text-emerald-600" /> Call Donor
                  </a>
                </div>

                {action && (
                  <button
                    onClick={() => onAdvanceStatus && onAdvanceStatus(task.id, action.next)}
                    className="w-full py-2.5 bg-emerald-600 hover:bg-emerald-700 text-white font-bold text-xs rounded-xl flex items-center justify-center gap-1.5 mt-2"
                  >
                    <Truck className="w-4 h-4" /> {action.label}
                  </button>
                )}
              </div>
            );
          })}
        </div>
      )}
    </div>
  );
}
