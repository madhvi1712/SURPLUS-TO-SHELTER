import React from 'react';
import { Recycle, AlertTriangle, Leaf } from 'lucide-react';

export default function RecycleHub({ items = [] }) {
  const expiredItems = items.filter(i => i.status === 'EXPIRED_ROUTED');

  return (
    <div className="space-y-4">
      <div className="border-b pb-3">
        <h2 className="text-xl font-bold text-slate-900">Municipal Bio-Energy & Recycling Loop</h2>
        <p className="text-xs text-slate-500">Rations exceeding safe consumption windows routed away from landfills into Biogas and Compost units.</p>
      </div>

      <div className="bg-amber-50 border border-amber-200 p-4 rounded-xl flex items-center gap-3 text-xs text-amber-900">
        <AlertTriangle className="w-5 h-5 text-amber-600 flex-shrink-0" />
        <span>Food Safety Rule Enforced: Batches shown here are locked from shelter human consumption to ensure health safety.</span>
      </div>

      {expiredItems.length === 0 ? (
        <div className="p-8 text-center bg-white rounded-2xl border text-slate-500 text-sm">
          No food batches currently diverted to bio-recycling. All items actively rescued within time windows!
        </div>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          {expiredItems.map(item => (
            <div key={item.id} className="bg-white p-5 rounded-2xl border border-slate-200 shadow-sm space-y-2">
              <div className="flex justify-between items-center">
                <span className="text-xs font-mono font-bold text-slate-700">Batch #{item.id}</span>
                <span className="text-[10px] font-bold px-2 py-0.5 rounded bg-slate-100 text-slate-700">Diverted from Landfill</span>
              </div>
              <h4 className="font-bold text-slate-900">{item.foodName} ({item.servings * 0.4} kg approx)</h4>
              <p className="text-xs text-slate-500">Origin: {item.donorName} ({item.pickupAddress})</p>
              <div className="text-xs text-emerald-700 font-semibold flex items-center gap-1 pt-2">
                <Leaf className="w-3.5 h-3.5" /> Destination: Regional Bio-CNG & Composting Plant
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}
